#include <tunables/global>

# The Supervisor renames the profile to the app's slug when it loads it.
profile cs_mailbox flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Files are left open: the image is the stock-Python cs-mailbox one, with the
  # server under /app, and cs-mailbox reads nothing outside it but /data, /ssl
  # and /run. The narrowing is in what the process may do, below.
  file,
  deny /ssl/** w,
  # The launcher execs /usr/local/bin/python3 on /app/mailbox/cs_mailbox.py;
  # the health check runs python3 too.
  /usr/local/bin/** ix,
  /usr/bin/** ix,
  /bin/** ix,

  # The launcher prepares /data as root, then becomes the mailbox user.
  capability chown,
  capability fowner,
  capability dac_override,
  capability dac_read_search,
  capability setuid,
  capability setgid,

  # A TCP server (and name lookups); no raw or packet sockets.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix,
  network netlink raw,

  signal (receive) set=(kill,term,int,hup,cont),
}
